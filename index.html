function checkScam(){ const messageInput = document.getElementById("scamText"); const result = document.getElementById("checkerResult"); if(!messageInput || !result){ alert("Scam Checker elements not found!"); return; } const message = messageInput.value.trim(); result.style.display = "block"; result.style.padding = "18px"; result.style.borderRadius = "10px"; result.style.lineHeight = "1.7"; // Check empty message if(message === ""){ result.style.background = "#fff1d8"; result.style.color = "#7a4b00"; result.textContent = currentLanguage === "fr" ? "⚠️ Veuillez entrer un message avant de commencer l'analyse." : "⚠️ Please enter a message before starting the analysis."; return; } // Normalize message const cleanMessage = message .toLowerCase() .normalize("NFD") .replace(/[\u0300-\u036f]/g, ""); // Suspicious expressions const suspiciousWords = [ // Money and Mobile Money scams "send money", "transfer money", "send your money", "mobile money pin", "share your pin", "momo pin", "give me your pin", "verification code", "send otp", "share otp", "urgent payment", "payment required", "refund your money", // Prize scams "you have won", "you won", "claim your prize", "free money", "cash prize", "congratulations winner", "you are selected", "claim reward", // Fake job scams "registration fee", "job application fee", "pay to get a job", "guaranteed employment", "job offer", "recruitment fee", "processing fee", // Investment scams "double your money", "guaranteed profit", "investment guaranteed", "quick profit", "earn money fast", // Phishing and account scams "click this link", "verify your account", "account blocked", "update your password", "confirm your password", "your account will be closed", // French suspicious expressions "envoyez de l'argent", "transferez votre argent", "partagez votre code pin", "code de verification", "vous avez gagne", "reclamez votre prix", "argent gratuit", "frais d'inscription", "frais de recrutement", "emploi garanti", "benefice garanti", "doublez votre argent", "cliquez sur ce lien", "verifiez votre compte", "compte bloque", "mot de passe" ]; // Detect suspicious expressions const detectedWords = suspiciousWords.filter(word => cleanMessage.includes( word.normalize("NFD").replace(/[\u0300-\u036f]/g, "") ) ); // Remove duplicates const uniqueWords = [...new Set(detectedWords)]; // Calculate warning level const riskScore = uniqueWords.length; let title = ""; let explanation = ""; if(riskScore >= 3){ result.style.background = "#ffe5e5"; result.style.color = "#9b1717"; title = currentLanguage === "fr" ? "🚨 RISQUE ÉLEVÉ : Plusieurs signes suspects détectés" : "🚨 HIGH RISK: Multiple suspicious signs detected"; explanation = currentLanguage === "fr" ? "Ce message contient plusieurs expressions associées aux arnaques. Ne transférez pas d'argent et ne partagez aucune information personnelle." : "This message contains several expressions associated with scams. Do not transfer money or share personal information."; }else if(riskScore > 0){ result.style.background = "#fff1d8"; result.style.color = "#805000"; title = currentLanguage === "fr" ? "⚠️ ATTENTION : Des signes suspects ont été détectés" : "⚠️ WARNING: Suspicious signs detected"; explanation = currentLanguage === "fr" ? "Ce message mérite une vérification supplémentaire. Ne prenez aucune décision financière avant de vérifier l'expéditeur." : "This message requires further checking. Do not make financial decisions before verifying the sender."; }else{ result.style.background = "#e8f5eb"; result.style.color = "#155b31"; title = currentLanguage === "fr" ? "🔎 Aucun mot suspect connu détecté" : "🔎 No known suspicious keywords detected"; explanation = currentLanguage === "fr" ? "Aucun des mots suspects enregistrés dans notre outil n'a été trouvé. Cela ne signifie pas que le message est sûr. Restez vigilant." : "None of the suspicious keywords in our tool were found. This does not mean the message is safe. Stay alert."; } // Display results result.replaceChildren(); const heading = document.createElement("h3"); heading.textContent = title; const description = document.createElement("p"); description.textContent = explanation; result.appendChild(heading); result.appendChild(description); // Show detected words if(uniqueWords.length > 0){ const detectedTitle = document.createElement("strong"); detectedTitle.textContent = currentLanguage === "fr" ? "Expressions suspectes trouvées :" : "Suspicious expressions found:"; const detectedList = document.createElement("ul"); uniqueWords.forEach(word => { const item = document.createElement("li"); item.textContent = word; detectedList.appendChild(item); }); result.appendChild(detectedTitle); result.appendChild(detectedList); } // Final safety advice const advice = document.createElement("p"); advice.textContent = currentLanguage === "fr" ? "🛡️ CONSEIL SAFELEARN : Ne partagez jamais votre code PIN, votre OTP ou votre mot de passe. Vérifiez toujours les informations auprès des services officiels." : "🛡️ SAFELEARN ADVICE: Never share your PIN, OTP or password. Always verify information through official services."; result.appendChild(advice); // Analysis disclaimer const disclaimer = document.createElement("small"); disclaimer.textContent = currentLanguage === "fr" ? "Note : cet outil utilise une détection par mots-clés. Il ne constitue pas une preuve qu'un message est frauduleux ou authentique." : "Note: This tool uses keyword detection. It does not prove that a message is fraudulent or genuine."; result.appendChild(disclaimer); }
